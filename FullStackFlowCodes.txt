@app.post("/predict", response_model=PredictionResponse)
async def predict(patient: PatientContext):
    features = extract_features(patient.clinical_text)
    probs = model.predict_proba(features)[0]
    pred_idx = np.argmax(probs)
    if probs[2] > THRESHOLD:
        pred_idx = 2
    class_map = {0: "NO", 1: ">30 days", 2: "<30 days"}
    return PredictionResponse(
        patient_id=patient.patient_id,
        risk_class=class_map[pred_idx],
        probabilities={"NO": probs[0], ">30 days": probs[1], "<30 days": probs[2]},
        risk_alert=probs[2] > THRESHOLD
    )

6. Slack Bot (slack_bot.py)
python
import os, re, requests
from slack_bolt import App
from slack_bolt.adapter.socket_mode import SocketModeHandler

SLACK_BOT_TOKEN = "xoxb-..."
SLACK_APP_TOKEN = "xapp-..."
API_URL = "http://52.15.208.206/predict"

app = App(token=SLACK_BOT_TOKEN)

@app.event("app_mention")
def handle_mention(event, say):
    clean_text = re.sub(r'<@[A-Z0-9]+>', '', event.get('text', '')).strip()
    if not clean_text:
        say("Please provide a patient context.")
        return
    if not re.match(r'^(predict|analyze|risk)\s+', clean_text, re.I):
        say("Use: @Readmission Predictor predict <patient context>")
        return
    patient_text = re.sub(r'^(predict|analyze|risk)\s+', '', clean_text, flags=re.I).strip()
    payload = {"patient_id": "slack_user", "clinical_text": patient_text}
    try:
        r = requests.post(API_URL, json=payload, timeout=30)
        if r.status_code == 200:
            data = r.json()
            result = f"{'🔴' if data['risk_class'] == '<30 days' else '🟡' if data['risk_class'] == '>30 days' else '🟢'} *Risk:* {data['risk_class']}\n{data['probabilities']}"
            say(result)
        else:
            say(f"API Error: {r.status_code}")
    except Exception as e:
        say(f"Error: {e}")

if __name__ == "__main__":
    handler = SocketModeHandler(app, SLACK_APP_TOKEN)
    handler.start()
